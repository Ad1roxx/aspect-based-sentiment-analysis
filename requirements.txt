# Install (from the repo root, with the venv active):
#     pip install -r requirements.txt

# Versions below are pinned to what actually resolved in the working venv on
# 2026-08-09, not to guesses — reproducibility is the whole point of pinning.
#
# NOTE ON TORCH: the pinned build is +cu126, the CUDA 12.6 wheel, which only
# exists on PyTorch's own package index (hence the --extra-index-url below).
# This is the *training* profile — it assumes an NVIDIA GPU.
#
# Serving does not need CUDA. When we containerise (build step 7) and set up CI
# (step 8), both run CPU-only, so those will need a CPU torch build instead.
# We split this file at that point rather than guessing the shape now.
# ---------------------------------------------------------------------------

--extra-index-url https://download.pytorch.org/whl/cu126

# --- Deep learning ---------------------------------------------------------
torch==2.13.0+cu126
transformers==5.14.1

# --- Data handling ---------------------------------------------------------
# XML parsing uses the standard library (xml.etree.ElementTree) — no dep needed.
numpy==2.4.6
# Downgraded from 3.0.5 when mlflow was added in Sprint 2: mlflow requires
# pandas<3. Recorded here rather than silently left stale — the pin has to match
# the venv it was resolved in or it is not a pin.
pandas==2.3.3

# --- Metrics ---------------------------------------------------------------
# f1_score / precision_recall_fscore_support / confusion_matrix. The train/val
# split is seeded-random rather than stratified — see the note in ml/src/data.py.
scikit-learn==1.9.0

# --- Experiment tracking (Sprint 2) ----------------------------------------
# Backend store is sqlite (ml/mlflow.db), not the classic ./mlruns file store.
# As of MLflow 3.x the file store is in maintenance mode and raises on use
# unless MLFLOW_ALLOW_FILE_STORE=true; it also never supported the Model
# Registry, which is what gives artifacts real version numbers. Verified, not
# assumed — see explanations/sprint-02.md.
mlflow==3.15.1
matplotlib==3.11.1    # confusion-matrix figures logged as run artifacts

# --- API serving (Sprint 4) -------------------------------------------------
# These arrive transitively via mlflow, but are pinned explicitly because the
# API depends on them directly. An undeclared dependency that happens to be
# installed is a dependency that breaks the day the package pulling it in is
# removed — and mlflow is exactly the package the serving container will drop.
fastapi==0.141.1
uvicorn==0.52.3
pydantic==2.13.4

# --- Utilities -------------------------------------------------------------
requests==2.34.2      # dataset download script
tqdm==4.70.0          # training progress bars
